<script>
// Product catalog (from database/schema.sql seed data)
const products = [
  { id: 1, name: 'Ashwagandha Capsules', sku: 'HP-ASH-060', description: 'Traditional Ayurvedic root extract for stress support, 60 capsules.', price: 349.00, stock: 120 },
  { id: 2, name: 'Triphala Churna', sku: 'HP-TRI-100', description: 'Classic three-fruit blend powder for digestion, 100g.', price: 199.00, stock: 80 },
  { id: 3, name: 'Chyawanprash', sku: 'HP-CHY-500', description: 'Herbal jam with amla and 40+ herbs, immunity support, 500g.', price: 425.00, stock: 60 },
  { id: 4, name: 'Brahmi Tablets', sku: 'HP-BRA-060', description: 'Bacopa monnieri tablets for focus and memory, 60 tablets.', price: 279.00, stock: 90 },
  { id: 5, name: 'Neem Face Wash', sku: 'HP-NFW-100', description: 'Purifying neem and tulsi face wash for clear skin, 100ml.', price: 149.00, stock: 150 },
  { id: 6, name: 'Turmeric Curcumin Capsules', sku: 'HP-TUR-060', description: 'High-potency curcumin with black pepper extract, 60 capsules.', price: 399.00, stock: 100 },
  { id: 7, name: 'Herbal Hair Oil', sku: 'HP-HHO-200', description: 'Bhringraj and amla oil blend for hair nourishment, 200ml.', price: 229.00, stock: 70 },
  { id: 8, name: 'Giloy Tablets', sku: 'HP-GIL-060', description: 'Guduchi extract tablets to support immunity, 60 tablets.', price: 259.00, stock: 85 },
  { id: 9, name: 'Amla Juice', sku: 'HP-AMJ-500', description: 'Cold-pressed Indian gooseberry juice, vitamin C rich, 500ml.', price: 189.00, stock: 65 },
  { id: 10, name: 'Shatavari Powder', sku: 'HP-SHA-100', description: "Asparagus racemosus powder for women's wellness, 100g.", price: 219.00, stock: 55 },
];

// cart = { productId: quantity }, in-memory only (resets on reload — a static
// page has no server/database to persist it, unlike the original PHP session/MySQL setup)
let cart = {};
let lastOrderId = 1000;

function money(n) { return n.toFixed(2); }

function showPage(name) {
  document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
  document.getElementById('page-' + name).classList.add('active');
  document.querySelectorAll('.nav a').forEach(a => a.classList.remove('active'));
  const navEl = document.querySelector(`.nav a[data-nav="${name}"]`);
  if (navEl) navEl.classList.add('active');
  if (name === 'shop') renderShop();
  if (name === 'cart') renderCart();
  if (name === 'checkout') renderCheckout();
}

function addToCart(id) {
  cart[id] = (cart[id] || 0) + 1;
  updateCartBadge();
}

function removeFromCart(id) {
  delete cart[id];
  updateCartBadge();
  renderCart();
}

function updateCartBadge() {
  const count = Object.values(cart).reduce((a, b) => a + b, 0);
  const badge = document.getElementById('cartBadge');
  if (count > 0) { badge.style.display = 'inline-block'; badge.textContent = count; }
  else { badge.style.display = 'none'; }
}

function productCard(p, extra) {
  return `<article class="card">
    <h3>${p.name}</h3>
    ${extra || `<p>${p.description}</p>`}
    <p class="price">₹${money(p.price)}</p>
    <button class="btn" onclick="addToCart(${p.id})">Add to Cart</button>
  </article>`;
}

function renderHome() {
  const latest = [...products].sort((a, b) => b.id - a.id).slice(0, 8);
  document.getElementById('homeGrid').innerHTML = latest.map(p => productCard(p)).join('');
}

function renderShop() {
  const q = (document.getElementById('searchInput').value || '').trim().toLowerCase();
  const list = q
    ? products.filter(p => p.name.toLowerCase().includes(q) || p.sku.toLowerCase().includes(q))
    : [...products].sort((a, b) => b.id - a.id);
  document.getElementById('shopGrid').innerHTML = list.map(p =>
    productCard(p, `<p>Stock: ${p.stock}</p>`)
  ).join('');
}

function renderCart() {
  const ids = Object.keys(cart);
  const container = document.getElementById('cartItems');
  const emptyMsg = document.getElementById('cartEmptyMsg');
  const checkoutBtn = document.getElementById('checkoutBtn');
  let total = 0;

  if (ids.length === 0) {
    container.innerHTML = '';
    emptyMsg.style.display = 'block';
    checkoutBtn.style.display = 'none';
  } else {
    emptyMsg.style.display = 'none';
    checkoutBtn.style.display = 'inline-block';
    container.innerHTML = ids.map(id => {
      const p = products.find(x => x.id === Number(id));
      const q = cart[id];
      total += p.price * q;
      return `<div class="card cart-line" style="margin-bottom:12px">
        <span><b>${p.name}</b> × ${q} — ₹${money(p.price * q)}</span>
        <a href="#" onclick="removeFromCart(${p.id});return false;">Remove</a>
      </div>`;
    }).join('');
  }
  document.getElementById('cartTotal').textContent = money(total);
}

function cartTotalValue() {
  return Object.entries(cart).reduce((sum, [id, q]) => {
    const p = products.find(x => x.id === Number(id));
    return sum + p.price * q;
  }, 0);
}

function renderCheckout() {
  if (Object.keys(cart).length === 0) { showPage('cart'); return; }
  document.getElementById('checkoutTotal').textContent = money(cartTotalValue());
  document.getElementById('checkoutError').style.display = 'none';
}

function placeOrder(e) {
  e.preventDefault();
  const name = document.getElementById('ckName').value.trim();
  const phone = document.getElementById('ckPhone').value.trim();
  const address = document.getElementById('ckAddress').value.trim();
  const errorEl = document.getElementById('checkoutError');

  if (!name || !phone || !address) {
    errorEl.textContent = 'Please fill in all fields.';
    errorEl.style.display = 'block';
    return;
  }

  // Simulate stock check + order insert (no real database in a static page)
  for (const [id, q] of Object.entries(cart)) {
    const p = products.find(x => x.id === Number(id));
    if (p.stock < q) {
      errorEl.textContent = 'Unable to place order. Please check stock and try again.';
      errorEl.style.display = 'block';
      return;
    }
  }
  Object.entries(cart).forEach(([id, q]) => {
    products.find(x => x.id === Number(id)).stock -= q;
  });

  lastOrderId += 1;
  document.getElementById('successMsg').textContent = `Order placed successfully. Order #${lastOrderId}`;
  cart = {};
  updateCartBadge();
  document.getElementById('ckName').value = '';
  document.getElementById('ckPhone').value = '';
  document.getElementById('ckAddress').value = '';
  showPage('success');
}

renderHome();
